cmake_minimum_required(VERSION 3.16)
project(PulseFeed LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "PulseFeed requires Linux")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PULSE_SANITIZER "" CACHE STRING "Optional sanitizer list, e.g. address,undefined or thread")
if(PULSE_SANITIZER)
    add_compile_options(-fsanitize=${PULSE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${PULSE_SANITIZER})
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
foreach(target pulsefeed publisher subscriber)
    if(target STREQUAL "pulsefeed")
        add_executable(pulsefeed src/relay.cpp)
    else()
        add_executable(${target} src/${target}.cpp)
    endif()
    target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(queue_test tests/queue_test.cpp)
    target_link_libraries(queue_test PRIVATE Threads::Threads)
    target_include_directories(queue_test PRIVATE src)
    add_test(NAME codec_and_queue COMMAND queue_test)
    add_test(NAME relay_integration
             COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py
                     $<TARGET_FILE:pulsefeed>)
    set_tests_properties(relay_integration PROPERTIES TIMEOUT 30)
endif()
